package challenge

type Decision string

const (
	Grant      Decision = "GRANT"
	Retain     Decision = "RETAIN"
	Revoke     Decision = "REVOKE"
	Quarantine Decision = "QUARANTINE"
)

type ChangeRequest struct {
	Subject    string
	Generation int
}

type Observation struct {
	Present    bool
	Verified   bool
	Subject    string
	Generation int
	Compliant  bool
}

// Accepts any observation at or after the requested generation.
func AuthorityDecision(req ChangeRequest, obs Observation) Decision {
	if req.Subject == "" || req.Generation < 1 {
		return Quarantine
	}
	if !obs.Present || !obs.Verified || obs.Subject != req.Subject || obs.Generation < req.Generation {
		return Quarantine
	}
	if obs.Compliant {
		return Retain
	}
	return Revoke
}
