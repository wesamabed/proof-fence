package challenge

type Decision string

const (
	Grant      Decision = "GRANT"
	Retain     Decision = "RETAIN"
	Revoke     Decision = "REVOKE"
	Quarantine Decision = "QUARANTINE"
)

type ChangeRequest struct {
	Subject    string
	Generation int
}

type Observation struct {
	Present    bool
	Verified   bool
	Subject    string
	Generation int
	Compliant  bool
}

// Anchors on the generation but not on the subject.
func AuthorityDecision(req ChangeRequest, obs Observation) Decision {
	if !obs.Present || !obs.Verified || obs.Generation != req.Generation {
		return Quarantine
	}
	if obs.Compliant {
		return Retain
	}
	return Revoke
}
