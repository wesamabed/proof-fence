package challenge

import "testing"

func req(subject string, generation int) ChangeRequest {
	return ChangeRequest{Subject: subject, Generation: generation}
}

func want(t *testing.T, got, expected Decision, what string) {
	t.Helper()
	if got != expected {
		t.Fatalf("%s: got %s, want %s", what, got, expected)
	}
}

func TestFreshMatchingCompliantObservation(t *testing.T) {
	obs := Observation{Present: true, Verified: true, Subject: "svc-a", Generation: 7, Compliant: true}
	want(t, AuthorityDecision(req("svc-a", 7), obs), Retain,
		"an attributable observation of this subject at this generation reporting compliance")
}

func TestFreshMatchingNonCompliantObservation(t *testing.T) {
	obs := Observation{Present: true, Verified: true, Subject: "svc-a", Generation: 7, Compliant: false}
	want(t, AuthorityDecision(req("svc-a", 7), obs), Revoke,
		"an attributable observation of this subject at this generation reporting non-compliance")
}

func TestGrantIsNeverCorrectHere(t *testing.T) {
	states := []Decision{
		AuthorityDecision(req("svc-a", 7), Observation{true, true, "svc-a", 7, true}),
		AuthorityDecision(req("svc-a", 7), Observation{true, true, "svc-a", 7, false}),
		AuthorityDecision(req("svc-a", 7), Observation{true, true, "svc-a", 6, true}),
		AuthorityDecision(req("svc-a", 7), Observation{}),
	}
	for i, d := range states {
		if d == Grant {
			t.Fatalf("state %d issued a grant to a subject that already holds one", i)
		}
	}
}
