package challenge

import "testing"
func TestExactAccepted(t *testing.T){ b,e:=ParseBinding([]byte(`{"run_id":"r1","authorized":true}`)); if e!=nil||b.RunID!="r1"||!b.Authorized{t.Fatalf("valid rejected %#v %v",b,e)} }
func TestAliasRejected(t *testing.T){ if _,e:=ParseBinding([]byte(`{"Run_ID":"r1","authorized":true}`)); e==nil{t.Fatal("case alias accepted")} }
func TestExactPlusAliasRejected(t *testing.T){ if _,e:=ParseBinding([]byte(`{"run_id":"r1","Run_ID":"r2","authorized":true}`)); e==nil{t.Fatal("alias decoy accepted")} }
