package challenge
import "testing"
func TestCoherentSuccess(t *testing.T){if !DeletionComplete(DeleteEvidence{Succeeded,"",true}){t.Fatal("coherent success rejected")}}
func TestDeniedCannotBeSuccess(t *testing.T){if DeletionComplete(DeleteEvidence{Denied,"AccessDenied",true}){t.Fatal("denied invocation promoted")}}

func TestMissingPositiveResponse(t *testing.T){if DeletionComplete(DeleteEvidence{Succeeded,"",false}){t.Fatal("missing response proof promoted")}}
func TestUnknownOutcome(t *testing.T){if DeletionComplete(DeleteEvidence{Outcome("MYSTERY"),"",true}){t.Fatal("unknown outcome promoted")}}
