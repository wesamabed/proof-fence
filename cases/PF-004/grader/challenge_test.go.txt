package challenge

import "testing"

func TestCoherentSuccessConfirms(t *testing.T) {
	if !RemovalConfirmed(RemovalResponse{Succeeded, "", true}) {
		t.Fatal("a coherent success envelope was rejected")
	}
}

func TestDeniedWithFault(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Denied, "AccessDenied", true}) {
		t.Fatal("a denied invocation was confirmed")
	}
}

func TestDeniedWithoutFault(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Denied, "", true}) {
		t.Fatal("a denied invocation with no recorded fault was confirmed")
	}
}

func TestFailedInvocation(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Failed, "", true}) {
		t.Fatal("a failed invocation was confirmed")
	}
}

func TestFailedWithFault(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Failed, "TimeoutError", true}) {
		t.Fatal("a failed invocation with a recorded fault was confirmed")
	}
}

func TestSucceededWithRecordedFault(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Succeeded, "ThrottlingException", true}) {
		t.Fatal("an envelope with a recorded fault was confirmed")
	}
}

func TestSucceededWithDifferentRecordedFault(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Succeeded, "InternalFailure", true}) {
		t.Fatal("an envelope with a recorded fault was confirmed")
	}
}

func TestSucceededWithoutPayloadConfirmation(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Succeeded, "", false}) {
		t.Fatal("an envelope whose payload does not report removal was confirmed")
	}
}

func TestUnmodelledResult(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Result("PARTIALLY_SUCCEEDED"), "", true}) {
		t.Fatal("a classification this client does not model was confirmed")
	}
}

func TestEmptyResult(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Result(""), "", true}) {
		t.Fatal("an envelope carrying no classification was confirmed")
	}
}

func TestSucceededFaultAndNoPayload(t *testing.T) {
	if RemovalConfirmed(RemovalResponse{Succeeded, "InternalFailure", false}) {
		t.Fatal("an envelope with a recorded fault and no payload confirmation was confirmed")
	}
}
