package challenge

type Result string

const (
	Succeeded Result = "SUCCEEDED"
	Denied    Result = "DENIED"
	Failed    Result = "FAILED"
)

type RemovalResponse struct {
	Result         Result
	FaultCode      string
	PayloadRemoved bool
}

// Ignores the independently recorded fault state.
func RemovalConfirmed(r RemovalResponse) bool {
	return r.Result == Succeeded && r.PayloadRemoved
}
