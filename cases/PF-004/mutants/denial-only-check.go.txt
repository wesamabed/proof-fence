package challenge

type Result string

const (
	Succeeded Result = "SUCCEEDED"
	Denied    Result = "DENIED"
	Failed    Result = "FAILED"
)

type RemovalResponse struct {
	Result         Result
	FaultCode      string
	PayloadRemoved bool
}

// Treats any classification that is not an explicit denial as a success.
func RemovalConfirmed(r RemovalResponse) bool {
	return r.Result != Denied && r.FaultCode == "" && r.PayloadRemoved
}
