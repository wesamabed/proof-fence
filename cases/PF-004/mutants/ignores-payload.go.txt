package challenge

type Result string

const (
	Succeeded Result = "SUCCEEDED"
	Denied    Result = "DENIED"
	Failed    Result = "FAILED"
)

type RemovalResponse struct {
	Result         Result
	FaultCode      string
	PayloadRemoved bool
}

// Never asks whether the payload reports removal.
func RemovalConfirmed(r RemovalResponse) bool {
	return r.Result == Succeeded && r.FaultCode == ""
}
