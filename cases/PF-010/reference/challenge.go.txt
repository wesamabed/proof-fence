package challenge

type ExecutionEvidence struct{ClaimedRun []string;ClaimedPass []string;TrustedObservedRun []string;TrustedObservedPass []string}
func ExecutionProven(required []string,e ExecutionEvidence)bool{run:=map[string]bool{};pass:=map[string]bool{};for _,x:=range e.TrustedObservedRun{run[x]=true};for _,x:=range e.TrustedObservedPass{pass[x]=true};for _,x:=range required{if !run[x]||!pass[x]{return false}};return true}
