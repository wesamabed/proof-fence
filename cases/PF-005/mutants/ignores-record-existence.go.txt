package challenge

type Finding struct {
	Determined bool
	Value      bool
}

type PathAnalysis struct {
	Recorded       bool
	SignatureValid bool
	ReportedNoPath bool
}

// Checks attribution but never asks whether a record exists at all.
func PathFinding(a PathAnalysis) Finding {
	if !a.SignatureValid {
		return Finding{}
	}
	return Finding{Determined: true, Value: a.ReportedNoPath}
}
