package challenge

type Finding struct {
	Determined bool
	Value      bool
}

type PathAnalysis struct {
	Recorded       bool
	SignatureValid bool
	ReportedNoPath bool
}

// Accepts any stored record without checking attribution.
func PathFinding(a PathAnalysis) Finding {
	if !a.Recorded {
		return Finding{}
	}
	return Finding{Determined: true, Value: a.ReportedNoPath}
}
