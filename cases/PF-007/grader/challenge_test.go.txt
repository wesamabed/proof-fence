package challenge

import "testing"

func TestAttributableProbeContained(t *testing.T) {
	if !ContainmentEstablished(ContainmentInputs{RequestField: true, ProbeRecorded: true, ProbeVerified: true, ProbeResult: true}) {
		t.Fatal("an attributable probe measuring containment was rejected")
	}
}

func TestProbeContainedDespiteRequestField(t *testing.T) {
	if !ContainmentEstablished(ContainmentInputs{RequestField: false, ProbeRecorded: true, ProbeVerified: true, ProbeResult: true}) {
		t.Fatal("an attributable probe measuring containment was rejected because the request field disagreed")
	}
}

func TestRequestFieldAlone(t *testing.T) {
	if ContainmentEstablished(ContainmentInputs{RequestField: true}) {
		t.Fatal("a caller-supplied request field established containment on its own")
	}
}

func TestRequestFieldOverridesProbe(t *testing.T) {
	if ContainmentEstablished(ContainmentInputs{RequestField: true, ProbeRecorded: true, ProbeVerified: true, ProbeResult: false}) {
		t.Fatal("a caller-supplied request field overrode an attributable probe measurement")
	}
}

func TestNoProbeRecord(t *testing.T) {
	if ContainmentEstablished(ContainmentInputs{RequestField: true, ProbeRecorded: false, ProbeVerified: true, ProbeResult: true}) {
		t.Fatal("containment was established with no probe record")
	}
}

func TestUnattributableProbe(t *testing.T) {
	if ContainmentEstablished(ContainmentInputs{RequestField: true, ProbeRecorded: true, ProbeVerified: false, ProbeResult: true}) {
		t.Fatal("an unattributable probe record established containment")
	}
}

func TestProbeMeasuredNotContained(t *testing.T) {
	if ContainmentEstablished(ContainmentInputs{RequestField: false, ProbeRecorded: true, ProbeVerified: true, ProbeResult: false}) {
		t.Fatal("a probe measuring no containment established containment")
	}
}

func TestNoEvidenceAtAll(t *testing.T) {
	if ContainmentEstablished(ContainmentInputs{}) {
		t.Fatal("containment was established with no evidence at all")
	}
}
