package challenge

type Attestation struct {
	SignatureValid bool
	Scope          string
	SubjectDigest  string
}

// Omits the check on the gate policy itself, so an unconfigured gate is
// satisfied by any scopeless attestation.
func AttestationSatisfies(requiredScope string, a Attestation) bool {
	return a.SignatureValid && a.Scope == requiredScope && a.SubjectDigest != ""
}
