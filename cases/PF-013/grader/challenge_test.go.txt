package challenge

import "testing"

func decide(intent Intent, status ApplyStatus, state SlotState) Decision {
	return RotationDecision(
		Request{Intent: intent},
		ApplyReport{Returned: true, Status: status},
		Readback{Present: true, Verified: true, State: state},
	)
}

func want(t *testing.T, got, expected Decision, what string) {
	t.Helper()
	if got != expected {
		t.Fatalf("%s: got %s, want %s", what, got, expected)
	}
}

func TestInstallCorroboratedByPresentKey(t *testing.T) {
	want(t, decide(Install, Applied, KeyPresent), Grant,
		"an install the service reports carried out, with the key read back in the slot")
}

func TestInstallContradictedByAbsentKey(t *testing.T) {
	want(t, decide(Install, Applied, KeyAbsent), Quarantine,
		"an install the service reports carried out, with no key read back in the slot")
}

func TestDeclinedInstallCorroboratedByAbsentKey(t *testing.T) {
	want(t, decide(Install, Rejected, KeyAbsent), Retain,
		"an install the service declined, with the slot read back still empty")
}

func TestDeclinedInstallContradictedByPresentKey(t *testing.T) {
	want(t, decide(Install, Rejected, KeyPresent), Quarantine,
		"an install the service declined, with a key nonetheless read back in the slot")
}

func TestRemoveCorroboratedByAbsentKey(t *testing.T) {

func TestUnattributableReadback(t *testing.T) {
	got := RotationDecision(
		Request{Intent: Install},
		ApplyReport{Returned: true, Status: Applied},
		Readback{Present: true, Verified: false, State: KeyPresent},
	)
	want(t, got, Quarantine, "a readback record whose signature did not verify")
}

func TestUnattributableReadbackOnRemoval(t *testing.T) {
	got := RotationDecision(
		Request{Intent: Remove},
		ApplyReport{Returned: true, Status: Applied},
		Readback{Present: true, Verified: false, State: KeyAbsent},
	)
	want(t, got, Quarantine, "a removal corroborated only by an unattributable readback")
}

func TestMissingReadbackRecord(t *testing.T) {
	got := RotationDecision(
		Request{Intent: Install},
		ApplyReport{Returned: true, Status: Applied},
		Readback{Present: false, Verified: true, State: KeyPresent},
	)
	want(t, got, Quarantine, "a readback record that was never loaded")
}

func TestNoResponseFromTheService(t *testing.T) {
	got := RotationDecision(
		Request{Intent: Install},

	)

}

func TestUnmodelledApplyStatus(t *testing.T) {
