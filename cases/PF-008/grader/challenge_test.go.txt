package challenge

import "testing"

func coherent() RetirementEvidence {
	return RetirementEvidence{
		Result:            Succeeded,
		FaultCode:         "",
		PayloadRetired:    true,
		InventoryRecorded: true,
		InventoryVerified: true,
		InventoryCount:    0,
	}
}

func TestCoherentRetirement(t *testing.T) {
	if !RetirementConfirmed(coherent()) {
		t.Fatal("coherent retirement evidence was rejected")
	}
}

func TestDeniedInvocation(t *testing.T) {
	e := coherent()
	e.Result = Denied
	e.FaultCode = "AccessDenied"
	if RetirementConfirmed(e) {
		t.Fatal("a denied registry invocation was confirmed")
	}
}

func TestFailedInvocation(t *testing.T) {
	e := coherent()
	e.Result = Failed
	if RetirementConfirmed(e) {
		t.Fatal("a failed registry invocation was confirmed")
	}
}

func TestUnmodelledResult(t *testing.T) {
	e := coherent()
	e.Result = Result("PARTIALLY_SUCCEEDED")
	if RetirementConfirmed(e) {
		t.Fatal("a classification this client does not model was confirmed")
	}
}
