package challenge

import (
    "bytes"
    "encoding/json"
    "errors"
    "fmt"
    "io"
)

type Grant struct { Subject string `json:"subject"`; Allow bool `json:"allow"` }

func ParseGrant(data []byte) (Grant, error) {
    dec := json.NewDecoder(bytes.NewReader(data))
    tok, err := dec.Token(); if err != nil || tok != json.Delim('{') { return Grant{}, errors.New("object required") }
    seen := map[string]bool{}; var g Grant
    for dec.More() {
        k, err := dec.Token(); if err != nil { return Grant{}, err }
        key, ok := k.(string); if !ok { return Grant{}, errors.New("string key required") }
        if seen[key] { return Grant{}, fmt.Errorf("duplicate member %q", key) }; seen[key]=true
        switch key {
        case "subject": if err := dec.Decode(&g.Subject); err != nil { return Grant{}, err }
        case "allow": if err := dec.Decode(&g.Allow); err != nil { return Grant{}, err }
        default: return Grant{}, fmt.Errorf("unknown member %q", key)
        }
    }
    if _, err := dec.Token(); err != nil { return Grant{}, err }
    if _, err := dec.Token(); err != io.EOF { if err == nil { return Grant{}, errors.New("trailing JSON") }; return Grant{}, err }
    if !seen["subject"] || !seen["allow"] || g.Subject=="" { return Grant{}, errors.New("missing required member") }
    return g,nil
}
